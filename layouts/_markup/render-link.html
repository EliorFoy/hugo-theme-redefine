{{- /* Link render hook — ports the external-link-icon Hexo filter: external links get class="link" and an arrow icon appended. */ -}} {{- $theme := site.Params -}} {{- $dest := .Destination -}} {{- $u := urls.Parse $dest -}} {{- $siteHost := urls.Parse (site.BaseURL) -}} {{- $external := false -}} {{- if and $u.Scheme (or (eq $u.Scheme "http") (eq $u.Scheme "https")) (ne $u.Host $siteHost.Host) -}} {{- $external = true -}} {{- end -}} {{- if and $external $theme.articles.style.link_icon -}} {{ .Text }} {{- else if $external -}} {{ .Text }} {{- else -}} {{ .Text }} {{- end -}}
