{{/* Parses a CSS size like "1000px" / "1.2rem" / 1000 into {value, unit}. Returns dict with "value" (float) and "unit" (string, "px" default). */}} {{- $s := string . -}} {{- $m := findRE "^(-?[0-9]*\\.?[0-9]+)" $s -}} {{- $value := 0.0 -}} {{- if $m }}{{ $value = float (index $m 0) }}{{ end -}} {{- $unit := replaceRE "^-?[0-9]*\\.?[0-9]+" "" $s -}} {{- if not $unit }}{{ $unit = "px" }}{{ end -}} {{- return (dict "value" $value "unit" $unit) -}}
