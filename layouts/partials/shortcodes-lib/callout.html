info :: Hello >}} ... {{< /callout >}} */ -}} {{- $positional := .sc.Params -}} {{- $isNamed := .sc.IsNamedParams -}} {{- $type := "default" -}} {{- $title := "" -}} {{- $iconClass := "" -}} {{- $extraClasses := slice -}} {{- $variant := "simple" -}} {{- if $isNamed -}} {{- $type = string (.sc.Get "type" | default "default") -}} {{- $title = string (.sc.Get "title" | default "") -}} {{- $iconClass = string (.sc.Get "icon" | default "") -}} {{- $variant = string (.sc.Get "variant" | default "simple") -}} {{- $cls := string (.sc.Get "class" | default (.sc.Get "classes")) -}} {{- range $c := (split $cls " ") }}{{ if $c }}{{ $extraClasses = $extraClasses | append $c }}{{ end }}{{ end -}} {{- if or (eq $variant "titled") (ne $title "") }}{{ $variant = "titled" }}{{ else }}{{ $variant = "simple" }}{{ end -}} {{- else -}} {{- /* positional: [type, ...icon/fa tokens, "::", title tokens...] */ -}} {{- $args := slice -}} {{- range $a := $positional }}{{ $args = $args | append (string $a) }}{{ end -}} {{- if gt (len $args) 0 }}{{ $type = index $args 0 }}{{ $args = $args | after 1 }}{{ end -}} {{- /* split at :: delimiter */ -}} {{- $delimIdx := -1 -}} {{- range $i, $a := $args }}{{ if and (eq $delimIdx -1) (eq $a "::") }}{{ $delimIdx = $i }}{{ end }}{{ end -}} {{- if ge $delimIdx 0 -}} {{- $titleParts := slice -}} {{- range $i, $a := $args }}{{ if gt $i $delimIdx }}{{ $titleParts = $titleParts | append $a }}{{ end }}{{ end -}} {{- $title = strings.TrimSpace (delimit $titleParts " ") -}} {{- $args = first $delimIdx $args -}} {{- $variant = "titled" -}} {{- end -}} {{- /* icon tokens: first non-style fa- token + optional style token */ -}} {{- $style := "" -}} {{- $icon := "" -}} {{- $rest := slice -}} {{- range $a := $args -}} {{- if and (strings.HasPrefix $a "fa-") (in (slice "fa-solid" "fa-regular" "fa-light" "fa-thin" "fa-duotone" "fa-brands" "fa-sharp" "fa-sharp-solid") $a) -}} {{- if not $style }}{{ $style = $a }}{{ end -}} {{- else if and (strings.HasPrefix $a "fa-") (not $icon) -}} {{- $icon = $a -}} {{- else -}} {{- $rest = $rest | append $a -}} {{- end -}} {{- end -}} {{- if $icon }}{{ $iconClass = printf "%s %s" (default "fa-solid" $style) $icon }}{{ end -}} {{- $extraClasses = $rest -}} {{- end -}} {{- $renderedContent := .sc.Page.RenderString .inner -}} {{- $iconMarkup := "" -}} {{- if $iconClass }}{{ $iconMarkup = printf `` $iconClass }}{{ end -}} {{- if eq $variant "titled" -}} {{- $renderedTitle := strings.TrimSpace (.sc.Page.RenderString (default "Note" $title)) -}} {{- $renderedTitle = replaceRE "^

([\\s\\S]*)

$" "$1" $renderedTitle -}} {{- $titleInner := $renderedTitle -}} {{- if $iconMarkup }}{{ $titleInner = printf "%s %s" $iconMarkup $renderedTitle }}{{ end -}}
{{ $titleInner | safeHTML }}
{{ $renderedContent | safeHTML }}
{{- else -}}
{{ $iconMarkup | safeHTML }}
{{ $renderedContent | safeHTML }}
