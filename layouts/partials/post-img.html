{{/* Builds the markup for article images (lazyload-aware). Input: dict { src, alt, title, lazy }. Returns an HTML string. */ -}} {{- $src := .src | safeURL -}} {{- $alt := .alt -}} {{- $titleAttr := "" -}} {{- with .title }}{{ $titleAttr = printf " title=\"%s\"" . }}{{ end -}} {{- $loading := "images/loading.svg" | relURL -}} {{- if .lazy -}} {{- return (printf `` $loading $src $alt $titleAttr) -}} {{- else -}} {{- return (printf `` $src $alt $titleAttr) -}} {{- end -}}
